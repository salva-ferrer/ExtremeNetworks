# install.txt - edge guard configuration snippet (EXOS / Switch Engine)
# Prerequisites: edge_guard.pol (or edge_guard-hw.pol saved on the switch as edge_guard.pol) and
# edge_guard.py already copied to the switch flash (see README.md, "Installation").
# Replace <access-ports>.

# 1) UPM profile. Interactive mode: after "create upm profile" paste the lines and finish
#    with a line that contains only a dot.
create upm profile edge_guard
set var p $TCL(regsub -all {^.* from [0-9]+:([0-9]+) .*$} ${EVENT.LOG_PARAM_0} {\1})
set var r $TCL(regsub -all {^.* matches rule ([A-Za-z0-9_]+):.*$} ${EVENT.LOG_PARAM_0} {\1})
load script edge_guard.py $p $r
.

# 2) EMS filter: only the Kern.Info messages generated by the edge_* policy entries
create log filter edge_f
configure log filter edge_f add events kern.info match string "matches rule edge_"

# 3) UPM log target that triggers the profile
create log target upm edge_guard
configure log target upm edge_guard filter edge_f
enable log target upm edge_guard

# 4) Check the policy and apply it on the ingress of the access ports
check policy edge_guard
configure access-list edge_guard ports <access-ports> ingress

# 5) Save once verified (see README.md, "Verification")
save
